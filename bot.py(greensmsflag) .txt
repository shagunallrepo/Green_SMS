def flag_for_number(value: str | None) -> str:
    """Return the ISO country flag for a phone number, when it is known."""
    if not value:
        return "🌐"
    try:
        # Clean number (remove spaces, +, -, etc.)
        clean_number = re.sub(r'\D', '', str(value))
        
        # Auto-detect region (don't force Pakistan)
        parsed = phonenumbers.parse(clean_number, None)
        
        if phonenumbers.is_valid_number(parsed):
            region = phonenumbers.region_code_for_number(parsed)
            if region in COUNTRY_FLAGS:
                return COUNTRY_FLAGS[region]
    except phonenumbers.phonenumberutil.NumberParseException:
        pass
    return "🌐"


def db_connection() -> sqlite3.Connection:
    connection = sqlite3.connect(DATABASE_FILE)
    connection.row_factory = sqlite3.Row
    connection.execute(
        """
        CREATE TABLE IF NOT EXISTS forwarded_messages (
            sms_id INTEGER PRIMARY KEY,
            claimed_at TEXT NOT NULL,
            sent_at TEXT
        )
        """
    )
    connection.execute(
        """
        CREATE TABLE IF NOT EXISTS bot_state (
            key TEXT PRIMARY KEY,
            value TEXT NOT NULL
        )
        """
    )
    connection.execute(
        """
        CREATE TABLE IF NOT EXISTS message_content (
            sms_id INTEGER PRIMARY KEY,
            text TEXT NOT NULL
        )
        """
    )
    connection.commit()
    return connection


def state_value(connection: sqlite3.Connection, key: str, default: str = "0") -> str:
    row = connection.execute(
        "SELECT value FROM bot_state WHERE key = ?", (key,)
    ).fetchone()
    return str(row["value"]) if row else default


def set_state(connection: sqlite3.Connection, key: str, value: str) -> None:
    connection.execute(
        """
        INSERT INTO bot_state(key, value) VALUES (?, ?)
        ON CONFLICT(key) DO UPDATE SET value = excluded.value
        """,
        (key, value),
    )
    connection.commit()


def claim_message(connection: sqlite3.Connection, sms_id: int) -> bool:
    """Claim an SMS unless it was already delivered successfully."""
    existing = connection.execute(
        "SELECT sent_at FROM forwarded_messages WHERE sms_id = ?", (sms_id,)
    ).fetchone()
    if existing and existing["sent_at"] is not None:
        return False
    if existing:
        return True
    cursor = connection.execute(
        """
        INSERT OR IGNORE INTO forwarded_messages(sms_id, claimed_at)
        VALUES (?, ?)
        """,
        (sms_id, datetime.now(timezone.utc).isoformat()),
    )
    connection.commit()
    return cursor.rowcount == 1


def mark_sent(connection: sqlite3.Connection, sms_id: int) -> None:
    connection.execute(
        "UPDATE forwarded_messages SET sent_at = ? WHERE sms_id = ?",
        (datetime.now(timezone.utc).isoformat(), sms_id),
    )
    connection.commit()


def save_message_content(
    connection: sqlite3.Connection, sms_id: int, text: str
) -> None:
    connection.execute(
        """
        INSERT INTO message_content(sms_id, text) VALUES (?, ?)
        ON CONFLICT(sms_id) DO UPDATE SET text = excluded.text
        """,
        (sms_id, text),
    )
    connection.commit()


def stored_message_content(
    connection: sqlite3.Connection, sms_id: int
) -> str | None:
    row = connection.execute(
        "SELECT text FROM message_content WHERE sms_id = ?", (sms_id,)
    ).fetchone()
    return str(row["text"]) if row else None


def sms_text(message: dict[str, Any]) -> str:
    """Extract sender number and message text from API response"""
    number = str(message.get("number", ""))
    flag = flag_for_number(number)
    
    # Mask number: +91•••6374 format
    if len(number) > 7:
        masked_number = f"{number[:3]}•••{number[-4:]}"
    else:
        masked_number = number
    
    text = str(message.get("sms", "(empty message)"))
    
    return f"{flag} {masked_number}\n{text}"


def otp_from_message(message_text: str) -> str:
    match = re.search(r"(?<!\d)\d{3,9}(?!\d)", message_text)
    return match.group(0) if match else "N/A"

    )

    rows = [row for row in (top_row, bottom_row) if row]
    return {"inline_keyboard": rows}


def fetch_sms(since_id: int) -> list[dict[str, Any]]:
    if not GREEN_SMS_API_KEY:
        raise RuntimeError("GREEN_SMS_API_KEY is not configured")
    response = requests.get(
        f"{GREEN_SMS_API}/messages/",
        headers={"Authorization": f"Bearer {GREEN_SMS_API_KEY}"},
        params={"since_id": since_id, "page_size": PAGE_SIZE},
        timeout=20,
    )
    response.raise_for_status()
    payload = response.json()
    results = payload.get("results", [])
    if not isinstance(results, list):
        raise RuntimeError("Green SMS response has an invalid results field")
    return sorted(
        [item for item in results if isinstance(item, dict) and item.get("id") is not None],
        key=lambda item: int(item["id"]),
    )


def telegram_call(method: str, payload: dict[str, Any]) -> dict[str, Any]:
    """Direct Telegram API call (LOCAL MACHINE VERSION)"""
    url = f"https://api.telegram.org/bot{BOT_TOKEN}/{method}"
    response = requests.post(url, json=payload, timeout=30)
    if not response.ok:
        raise RuntimeError(f"Telegram HTTP {response.status_code}: {response.text}")
    result = response.json()
    if not result.get("ok"):
        raise RuntimeError(f"Telegram rejected request: {result.get('description')}")
    return result

def telegram_updates(offset: int) -> list[dict[str, Any]]:
    """Get updates from Telegram"""
    url = f"https://api.telegram.org/bot{BOT_TOKEN}/getUpdates"
    params = {"offset": offset, "timeout": 1, "limit": 100}
    response = requests.get(url, params=params, timeout=30)
    if not response.ok:
        raise RuntimeError(f"Telegram updates HTTP {response.status_code}: {response.text}")
    result = response.json()
    if not result.get("ok"):
        raise RuntimeError(f"Telegram updates failed: {result.get('description')}")
    return result.get("result", [])

        f"Last poll: {runtime_last_poll or 'not started'}\n"
        f"Last error: {last_error or 'none'}"
    )


async def forward_loop(connection: sqlite3.Connection) -> str | None:
    global runtime_last_error, runtime_last_poll
    last_error: str | None = None
    while True:
        try:
            runtime_last_poll = datetime.now(timezone.utc).isoformat()
            cursor = int(state_value(connection, "last_sms_id"))
            for message in fetch_sms(cursor):
                sms_id = int(message["id"])
                if not claim_message(connection, sms_id):
                    set_state(connection, "last_sms_id", str(max(cursor, sms_id)))
                    continue
                full_text = sms_text(message)
                save_message_content(connection, sms_id, full_text)
                telegram_call(
                    "sendMessage",
                    {
                        "chat_id": TELEGRAM_GROUP_ID,
                        "text": full_text,
                        "reply_markup": message_buttons(full_text, sms_id),
                    },
                )
                mark_sent(connection, sms_id)
                set_state(connection, "last_sms_id", str(max(cursor, sms_id)))
            last_error = None
            runtime_last_error = None
        except Exception as error:
            last_error = str(error)
            runtime_last_error = last_error
            print(f"Forwarding error: {last_error}", flush=True)
        await asyncio.sleep(POLL_SECONDS)


async def command_loop(connection: sqlite3.Connection) -> None:

    offset = int(state_value(connection, "telegram_update_offset"))
    while True:
        try:
            updates = telegram_updates(offset)
            for update in updates:
                update_id = int(update.get("update_id", 0))
                offset = max(offset, update_id + 1)
                set_state(connection, "telegram_update_offset", str(offset))
                message = update.get("message", {})
                chat = message.get("chat", {})
                callback = update.get("callback_query")
                sender = (
                    callback.get("from", {})

                    else message.get("from", {})
                )
                text = str(message.get("text", ""))
                
                if chat.get("type") != "private":
                    if not callback:
                        continue
                        
                if callback:
                    data = str(callback.get("data", ""))

                        except ValueError:

                            "Full message is no longer available."

                            "answerCallbackQuery",

                                "callback_query_id": callback["id"],
                                "text": content[:200],
                                "show_alert": True,

                        )
                        if len(content) > 200:
                            telegram_call(
                                "sendMessage",
                                {
                                    "chat_id": sender["id"],
                                    "text": f"Full SMS (ID {sms_id}):\n{content}",

                            )
                    continue
                    
                # Handle commands only for owner

                    cmd = text.strip().split(maxsplit=1)[0].split("@")[0].lower()
                    
                    if cmd == "/start":

                            "sendMessage",

                                "chat_id": chat["id"],

        except Exception as error:
